cmake_minimum_required(VERSION 3.25)
project(radiance_audit_collector LANGUAGES CXX)
find_package(JNI REQUIRED)
set(MCVR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../MCVR" CACHE PATH "Matched MCVR checkout")
if(NOT EXISTS "${MCVR_ROOT}/src/core/diagnostics/audit_api.h")
    message(FATAL_ERROR "Set MCVR_ROOT to the matched checkout")
endif()
add_library(radiance-audit-native SHARED collector.cpp)
target_compile_features(radiance-audit-native PRIVATE cxx_std_23)
target_include_directories(radiance-audit-native PRIVATE ${JNI_INCLUDE_DIRS} "${MCVR_ROOT}/src/core/diagnostics")
target_compile_definitions(radiance-audit-native PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
enable_testing()
add_executable(audit_collector_test collector_test.cpp)
target_compile_features(audit_collector_test PRIVATE cxx_std_23)
target_include_directories(audit_collector_test PRIVATE "${MCVR_ROOT}/src/core/diagnostics")
add_test(NAME audit.native-collector COMMAND audit_collector_test)
add_executable(audit_profile_collector_test profile_collector_test.cpp)
target_compile_features(audit_profile_collector_test PRIVATE cxx_std_23)
target_include_directories(audit_profile_collector_test PRIVATE "${MCVR_ROOT}/src/core/diagnostics")
add_test(NAME audit.profile-collector COMMAND audit_profile_collector_test)

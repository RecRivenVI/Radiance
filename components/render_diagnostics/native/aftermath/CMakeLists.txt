cmake_minimum_required(VERSION 3.24)
project(radiance_local_aftermath LANGUAGES CXX)
option(RADIANCE_CAPTURE_AGENT "Build the local SDK-dependent JVM agent" ON)
set(AFTERMATH_SDK "" CACHE PATH "Private SDK input; never copied to the mod")
if(RADIANCE_CAPTURE_AGENT)
find_package(JNI REQUIRED)
find_package(Vulkan REQUIRED)
if(NOT EXISTS "${AFTERMATH_SDK}/include/GFSDK_Aftermath_GpuCrashDump.h")
    message(FATAL_ERROR "Specify the installed local Aftermath SDK")
endif()
configure_file(sdk_path.hpp.in sdk_path.hpp @ONLY)
add_library(radiance-aftermath-agent SHARED agent.cpp)
target_compile_features(radiance-aftermath-agent PRIVATE cxx_std_20)
target_compile_definitions(radiance-aftermath-agent PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(radiance-aftermath-agent PRIVATE /utf-8)
target_include_directories(radiance-aftermath-agent PRIVATE
    "${AFTERMATH_SDK}/include" ${JNI_INCLUDE_DIRS} "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(radiance-aftermath-agent PRIVATE Vulkan::Headers)
# Dynamic load of the existing absolute SDK DLL. No link, copy or install rule.
endif()
enable_testing()
add_executable(capture_wait_test wait_test.cpp)
target_compile_features(capture_wait_test PRIVATE cxx_std_20)
add_test(NAME local-capture.bounded-wait COMMAND capture_wait_test)
add_executable(capture_output_test output_test.cpp)
target_compile_features(capture_output_test PRIVATE cxx_std_20)
add_test(NAME local-capture.output COMMAND capture_output_test)
